```bash
node -e "
const c=require('crypto'),A='ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
for(let i=0;i<100;i++){
  const g=()=>Array.from({length:4},()=>A[c.randomInt(A.length)]).join('');
  const code='VIP-'+g()+'-'+g();
  console.log(code, c.createHash('sha256').update('shkoladrop-vip:'+code).digest('hex').slice(0,24));

```
